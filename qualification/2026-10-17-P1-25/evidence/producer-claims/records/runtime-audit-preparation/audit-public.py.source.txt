def pin(p):
 p=Path(p);d=p.read_bytes();return {'path':str(p),'bytes':len(d),'sha256':hashlib.sha256(d).hexdigest()}

def timestamp_ms(value):
 d=datetime.datetime.fromisoformat(value.replace('Z','+00:00'))
 assert d.tzinfo is not None
 delta=d.astimezone(datetime.timezone.utc)-datetime.datetime(1970,1,1,tzinfo=datetime.timezone.utc)
 assert delta.microseconds%1000==0
 return delta.days*86_400_000+delta.seconds*1000+delta.microseconds//1000

def claimed_iso(value):
 assert type(value) is int
 return (datetime.datetime(1970,1,1,tzinfo=datetime.timezone.utc)+datetime.timedelta(milliseconds=value)).isoformat(timespec='milliseconds').replace('+00:00','Z')

def accuracy(value):
 if value is None:return Decimal(0)
 assert type(value) in (int,float)
 n=Decimal(str(value));assert n.is_finite() and n>=0
 return n

def attested_matches(value,grant):

 assert hashlib.sha256(base64.b64decode(value['tokenDerB64'],validate=True)).hexdigest()==grant['tokenSha256']

def cadence_file(row,expected,mode=None):

 return current
