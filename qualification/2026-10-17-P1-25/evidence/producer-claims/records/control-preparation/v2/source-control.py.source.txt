def sha(data):
    return hashlib.sha256(data).hexdigest()


def load(path):
    return json.loads(Path(path).read_text())


def pin(path):
    path = Path(path)
    st = path.lstat()
    assert stat.S_ISREG(st.st_mode) and st.st_nlink == 1 and path.resolve() == path, str(path)
    data = path.read_bytes()
    end = path.lstat()
    assert (st.st_dev, st.st_ino, st.st_nlink, st.st_size, st.st_mtime_ns) == (end.st_dev, end.st_ino, end.st_nlink, end.st_size, end.st_mtime_ns), str(path)
    return {'path': str(path), 'physicalPath': str(path.resolve()), 'bytes': len(data), 'sha256': sha(data),
            'dev': st.st_dev, 'inode': st.st_ino, 'nlink': st.st_nlink}, data


def verify(row, physical=False):
    actual, data = pin(row['path'])
    assert (actual['bytes'], actual['sha256']) == (row['bytes'], row['sha256']), row['path']
    if physical:
        assert all(actual[k] == row[k] for k in ('physicalPath', 'dev', 'inode', 'nlink')), row['path']
    return actual, data


def git(root, *args):

def preservation(binding, protected, q, mutant=None, target_relative=None):
    seen = set()
    for row in binding['boundPaths']:
        rel = row['relativePath']
        assert not Path(rel).is_absolute() and '..' not in Path(rel).parts
        for key, root in (('source', S), ('fresh', q)):
            expected = row[key]
            assert expected['path'] == str(root / rel)
            actual, data = pin(root / rel)
            assert all(actual[k] == expected[k] for k in ('physicalPath', 'dev', 'inode', 'nlink'))
            assert (actual['dev'], actual['inode']) not in seen
            seen.add((actual['dev'], actual['inode']))
            digest, size = (mutant['sha256'], mutant['bytes']) if mutant and key == 'fresh' and rel == target_relative else (row['sha256'], row['bytes'])
            assert (actual['sha256'], actual['bytes']) == (digest, size), str(root / rel)
    for row in protected['inputs']:
        verify(row, True)
    natives = [*protected['nativePins'], binding['freshNative']]
    assert binding['freshNative']['path'] == str(q / 'node_modules/better-sqlite3/build/Release/better_sqlite3.node')
    native_ids = []
    for row in natives:
        actual, unused = verify(row, True)
        native_ids.append((actual['dev'], actual['inode']))
    assert len(set(native_ids)) == len(natives) == len(protected['nativePins']) + 1 and not (set(native_ids) & seen)
    return {'selectedPathsEachRoot': len(binding['boundPaths']), 'compiledPairs': len(binding['compiledPairs']),
            'protectedInputs': len(protected['inputs']), 'nativePhysicalPins': len(natives),
            'allSelectedBytesAndPhysicalPinsExact': True, 'heldRHeadStatusIndexExact': True,
            'runtimeProtectedInputsPin': pin(B / 'runtime-protected-inputs.json')[0],
            'prechangeProtectedInputsPin': pinBASE_PROTECTED[0],
            'oldOneTestSourcePreservedAndOnlyOneExplicitSuccessor': True}

    assert out['groupAbsentByESRCH'] is True and out['signalsOtherThanZeroSent'] is False
    assert command['cwd'] == str(R) and command['head'] == command['headAfter'] == load(B / 'runtime-protected-inputs.json')['heldRHead']

    assert command['cmd'] == out['command']['cmd'] and command['durationMs'] == out['command']['durationMs']
    argv = command['ownedProcess']['argv']
    assert argv[argv.index('--root') + 1] == str(root)
    assert tuple(arg for arg in argv if arg in CORPUS) == CORPUS
    for role in ('rawProducer', 'rawSnapshot', 'observer'):
        verify(command['artifacts'][role])
    assert out['rawObserverOrderedJoin'] is True and not out['unhandledErrors'] and not out['lateDiagnostics']
    total = out['tests']['total']
    assert isinstance(total, int) and total > 0 and out['tests']['skipped'] == 0
    assert out['tests']['passed'] + out['tests']['failed'] == total
    assert len(out['orderedIdentities']) == len({tuple(row) for row in out['orderedIdentities']}) == total

    assert len(out['taskErrors']) == out['tests']['failed']
    assert isinstance(out['hookTaskEntries'], int) and isinstance(out['individualHookValues'], int)
    return out, outcome_pin, command_pin

def action_closed(number, mode, operation, dependent_phase):
    # Root records the actual direct helper return only after natural closure; never a future hash.
    action_path = B / ('control-action-' + mode + number + '.json')
    action_pin, action_bytes = pin(action_path)
    action = json.loads(action_bytes)
    assert action['exit'] == 0 and action['stderr'] == ''
    assert action['argv'][-3:] == [str(B / 'source-control.py'), mode, number]
    assert '-B' in action['argv'][:-3]
    assert json.loads(action['stdout']) == operation
    started = datetime.datetime.fromisoformat(action['startedUtc'])
    ended = datetime.datetime.fromisoformat(action['endedUtc'])
    recorded = datetime.datetime.fromisoformat(operation['at'])
    assert all(value.tzinfo is not None for value in (started, ended, recorded))
    assert started <= recorded <= ended
    dependent = load(N / ('command-' + dependent_phase) / 'command.json')
    assert dependent['phase'] == dependent_phase and dependent['endedUtc']
    assert dependent['groupAbsentByESRCH'] is True and dependent['signalsOtherThanZeroSent'] is False
    assert ended <= datetime.datetime.fromisoformat(dependent['startedUtc'])
    return action_pin


def write_json(path, value):
    with path.open('x') as handle:
        json.dump(value, handle, indent=2)
        handle.write('\n')


def guarded_write(target, expected, replacement, source, root_binding_pin, prior_action_pin=None):
    source_pin, unused = pin(source)
    before, data = pin(target)
    assert data == expected and (before['dev'], before['inode']) != (source_pin['dev'], source_pin['inode'])
    fd = os.open(target, os.O_RDWR | os.O_NOFOLLOW)
    with os.fdopen(fd, 'r+b') as handle:
        st = os.fstat(handle.fileno())
        assert stat.S_ISREG(st.st_mode) and st.st_nlink == 1 and (st.st_dev, st.st_ino) == (before['dev'], before['inode'])
        assert handle.read() == expected
        again = target.lstat()
        assert (again.st_dev, again.st_ino, again.st_nlink) == (st.st_dev, st.st_ino, 1) and target.resolve() == target
        inactive()
        verify(root_binding_pin, True)
        if prior_action_pin is not None:
            verify(prior_action_pin, True)
        handle.seek(0)
        handle.truncate(0)
        assert handle.write(replacement) == len(replacement)
        handle.flush()
        os.fsync(handle.fileno())
    after, actual = pin(target)
    assert actual == replacement and (after['dev'], after['inode']) == (before['dev'], before['inode'])
    return before, after
