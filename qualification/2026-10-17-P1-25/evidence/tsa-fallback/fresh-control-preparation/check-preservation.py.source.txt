from pathlib import Path
import datetime
import importlib.util
import json
import sys

P = Path('/private/tmp/p1-25-fallback-qualification-20261017/fresh-control-preparation')
spec = importlib.util.spec_from_file_location('p125f_control_support', P / 'source-control.py')
support = importlib.util.module_from_spec(spec)
spec.loader.exec_module(support)
label, = sys.argv[1:]
assert label in ('before-controls', 'after-m1', 'after-m2', 'final')
binding, protected, q = support.context()
assert support.git(q, 'status', '--porcelain=v1') == b''

source, unused = support.pin(support.S / support.TARGET)

       'at': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'label': label,
       'sourceCommit': binding['sourceCommit'], 'sourceRoot': str(support.S), 'freshRoot': str(q),
       **facts, 'target': target, 'sourceTarget': source,
       'bothTreesClean': True, 'bothIndexesEmpty': True,
       'boundary': 'Actual read-only boundary snapshot only. No continuous isolation, unrecorded earlier check, runtime or future restoration credit.'}
support.write_json(support.B / ('preservation-' + label + '.json'), out)
print(json.dumps(out))
