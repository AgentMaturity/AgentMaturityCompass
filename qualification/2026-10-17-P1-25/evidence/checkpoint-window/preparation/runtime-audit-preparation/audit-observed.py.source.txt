"""Reuse the sealed P1-25 raw/observer join without altering its evidence or checks."""
from pathlib import Path
import hashlib
import importlib.util
import json
import re
import sys

B = Path('/private/tmp/p1-25-checkpoint-qualification-20261017')
HELPER = Path('/private/tmp/p1-25-time-qualification-20261016/receipt-preparation/assemble-receipt.py')
assert hashlib.sha256(HELPER.read_bytes()).hexdigest() == '28ddcb18ec76480886dc6856dee2a976adf15d2282c1b72d58c26ba22269a5c7'
spec = importlib.util.spec_from_file_location('prior_qualification_evidence', HELPER)
helper = importlib.util.module_from_spec(spec)
spec.loader.exec_module(helper)
phase, root, cohort = sys.argv[1:]
root_path = Path(root)
assert root_path.is_absolute() and root_path.is_dir() and root_path.resolve() == root_path
unchanged = ('tests/time/verifyTimestamp.test.ts',)
assert cohort in ('unchanged', 'combined')
helper.CORPUS = unchanged if cohort == 'unchanged' else ('tests/time/checkpointWindow.test.ts', 'tests/time/tsaGrant.test.ts', *unchanged)
facts, publics = helper.observed(phase, root_path)
grant_publics = re.findall(r'^P125_TSA_GRANT_EVIDENCE (/.+public-evidence\.json)$', (helper.N / ('command-' + phase) / 'stdout-stderr.log').read_text(), re.M)
checkpoint_publics = re.findall(r'^P125_CHECKPOINT_EVIDENCE (/.+checkpoint-public-evidence\.json)$', (helper.N / ('command-' + phase) / 'stdout-stderr.log').read_text(), re.M)
facts['publicEvidencePaths'] = [str(path) for path in publics] + grant_publics + checkpoint_publics
facts['grantPublicEvidencePaths'] = grant_publics
facts['checkpointPublicEvidencePaths'] = checkpoint_publics
facts['reuseBoundary'] = 'Read-only import of the pinned existing ordered raw/observer/hook/late-error audit; only the finite cohort and literal public path markers are selected here; no public path is dereferenced. No test execution or evidence rewriting.'
with (B / (phase + '-outcome.json')).open('x') as out:
    json.dump(facts, out, indent=2)
    out.write('\n')
print(json.dumps({key: facts[key] for key in ('phase', 'tests', 'hookTaskEntries', 'individualHookValues', 'rawObserverOrderedJoin', 'unhandledErrors', 'lateDiagnostics')}))
