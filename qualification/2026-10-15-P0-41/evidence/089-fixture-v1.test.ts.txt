function sha(bytes: string | Buffer): string { return createHash("sha256").update(bytes).digest("hex"); }

function inside(parent: string, child: string): boolean {
  const suffix = relative(parent, child);
  return suffix === "" || !isAbsolute(suffix) && suffix !== ".." && !suffix.startsWith(".." + sep);
}
function retainedRoot(): string | undefined {
  const selected = process.env[ARTIFACT_ENV];
  if (selected === undefined) return undefined;
  if (!isAbsolute(selected) || lstatSync(selected).isSymbolicLink() || !lstatSync(selected).isDirectory()) {
    throw new Error(ARTIFACT_ENV + " must name an existing absolute, non-symlink private directory");
  }
  const root = realpathSync(selected);
  if (inside(REPO, root) || inside(root, REPO)) throw new Error("Artifact directory must be separate from source");
  return root;
}
function wxJson(path: string, value: unknown): void {
  const encoded = JSON.stringify(value, (_key, item: unknown) => {
    if (item === undefined) return { p041NonJsonValue: "undefined" };
    if (typeof item === "number" && !Number.isFinite(item)) return { p041NonJsonValue: String(item) };
    return item;
  }, 2);
  writeFileSync(path, encoded + "\n", { flag: "wx", mode: 0o600 });
}

function put(fixture: Fixture, path: string, bytes: string | Buffer): void {
  const target = join(fixture.root, path);
  mkdirSync(dirname(target), { recursive: true });

  fixture.extraPaths.add(path);
}
function putJson(fixture: Fixture, path: string, value: unknown): void { put(fixture, path, JSON.stringify(value, null, 2) + "\n"); }
function snapshot(fixture: Fixture, directory: string): unknown[] {
  return [...new Set([...INPUT_PATHS, ...fixture.extraPaths])].sort().map(path => {
    const target = join(fixture.root, path);
    if (!existsSync(target)) return { path, exists: false };
    const bytes = readFileSync(target);
    const copy = join(directory, "inputs", path);
    mkdirSync(dirname(copy), { recursive: true });
    writeFileSync(copy, bytes, { flag: "wx", mode: 0o600 });
    return { path, exists: true, bytes: bytes.length, sha256: sha(bytes) };
  });
}
function command(fixture: Fixture, executable: string, args: string[]): Command {
  fixture.sequence += 1;
  const directory = fixture.evidence ? join(fixture.evidence, String(fixture.sequence).padStart(3, "0")) : undefined;
  if (directory) {
    mkdirSync(directory);
    const inputs = snapshot(fixture, directory);
    wxJson(join(directory, "intent.json"), { argv: [executable, ...args], cwd: fixture.root,
      envKeys: Object.keys(fixture.env).sort(), commits: fixture.commits, inputs,
      note: "No environment values retained; no command timeout/retry/termination added." });
  }
  // File-backed streams avoid spawnSync's pipe-buffer termination path.
  // Synchronous normal completion only; no new timeout, signal or retry.
  const streams = directory ?? join(fixture.temporary, String(fixture.sequence).padStart(3, "0"));
  if (!directory) mkdirSync(streams);
  const stdoutPath = join(streams, "stdout.log"), stderrPath = join(streams, "stderr.log");
  const child = (() => {
    const stdout = openSync(stdoutPath, "wx", 0o600);
    try {
      const stderr = openSync(stderrPath, "wx", 0o600);
      try { return spawnSync(executable, args, { cwd: fixture.root, env: fixture.env, stdio: ["ignore", stdout, stderr] }); }
      finally { closeSync(stderr); }
    } finally { closeSync(stdout); }
  })();
  const result: Command = { argv: [executable, ...args], cwd: fixture.root, pid: child.pid ?? null,
    status: child.status, signal: child.signal, stdout: readFileSync(stdoutPath, "utf8"), stderr: readFileSync(stderrPath, "utf8"),
    error: child.error ? { name: child.error.name, message: child.error.message,
      code: "code" in child.error ? String(child.error.code) : undefined } : null };
  if (directory) wxJson(join(directory, "result.json"), result);
  return result;
}
function git(fixture: Fixture, args: string[]): string {
  const result = command(fixture, "git", args);
  if (result.error || result.signal || result.status !== 0) {
    throw new Error("Owned fixture Git command failed; inspect retained command record: " + args.join(" "));
  }
  return result.stdout.trim();
}
function commit(fixture: Fixture, paths: string[], message: string): string {
  git(fixture, ["add", "--", ...paths]);
  git(fixture, ["commit", "-m", message]);
  const id = git(fixture, ["rev-parse", "--verify", "HEAD"]);
  if (!/^[a-f0-9]{40}$/.test(id)) throw new Error("Fixture did not yield a genuine full commit ID");
  fixture.commits.push(id);
  if (fixture.evidence) wxJson(join(fixture.evidence, "commit-" + fixture.commits.length + ".json"), {
    commit: id, parents: git(fixture, ["show", "-s", "--format=%P", id]), scope: "genuine local synthetic commit; no merge approval or signature assertion"
  });
  return id;
}
function cli(fixture: Fixture, args: string[]): Command {
  return command(fixture, process.execPath, [join(fixture.root, "scripts/program-metrics.mjs"), ...args]);
}
function success(result: Command): void {
  expect(result.error).toBeNull(); expect(result.signal).toBeNull(); expect(result.status).toBe(0); expect(result.stderr).toBe("");
}
function refusal(result: Command, diagnostic: string): void {
  expect(result.error).toBeNull(); expect(result.signal).toBeNull(); expect(result.status).toBe(1);
  expect(result.stdout).toBe(""); expect(result.stderr).toContain("program-metrics: " + diagnostic);
}
function fixture(id: string): Fixture {
  const selected = retainedRoot();
  const owned = mkdtempSync(join(selected ?? tmpdir(), id + "-p041-"));
  const root = join(owned, "fixture"), home = join(owned, "home"), temporary = join(owned, "tmp");
  mkdirSync(root); mkdirSync(home); mkdirSync(temporary);
  const evidence = selected ? join(owned, "evidence") : undefined;
  if (evidence) mkdirSync(evidence);
  const globalConfig = join(home, "gitconfig");
  writeFileSync(globalConfig, "", { flag: "wx", mode: 0o600 });
  const env: NodeJS.ProcessEnv = { PATH: process.env.PATH, HOME: home, TMPDIR: temporary, LANG: "C", LC_ALL: "C",
    GIT_CONFIG_GLOBAL: globalConfig, GIT_CONFIG_NOSYSTEM: "1", GIT_TERMINAL_PROMPT: "0" };
  const selectedFixture: Fixture = { root, home, temporary, evidence, env, sequence: 0, extraPaths: new Set(), commits: [] };
  for (const path of INPUT_PATHS) put(selectedFixture, path, readFileSync(join(REPO, path)));
  // Borrow installed YAML resolution only. Never stage the link or run dependency commands.
  symlinkSync(join(REPO, "node_modules"), join(root, "node_modules"), "dir");
  if (evidence) wxJson(join(evidence, "fixture.json"), { id, root, home, temporary,
    envKeys: Object.keys(env).sort(), scriptBytes: readFileSync(join(root, INPUT_PATHS[0]!)).length,
    scriptSha256: sha(readFileSync(join(root, INPUT_PATHS[0]!))), borrowedModules: realpathSync(join(REPO, "node_modules")),
    preserved: true, artifactMode: "required by later qualification runner; optional for ordinary test invocation" });
  git(selectedFixture, ["init"]);
  git(selectedFixture, ["config", "--local", "user.name", "P041 Synthetic Fixture"]);
  git(selectedFixture, ["config", "--local", "user.email", "p041-fixture@example.invalid"]);
  commit(selectedFixture, INPUT_PATHS, "Initial synthetic P041 fixture");
  return selectedFixture;
}

  });
