function fromBase64(value: string): string {
  return Buffer.from(value, "base64").toString("hex");
}

// Independent, naive RFC construction over raw bytes. It uses neither production
// hashing helpers nor production roots/proofs/frontiers. These inputs are synthetic.
function referenceRoot(data: readonly Uint8Array[]): string {
  if (data.length === 0) return createHash("sha256").digest("hex");
  if (data.length === 1) {
    return createHash("sha256").update(Buffer.concat([Buffer.from([0]), Buffer.from(data[0]!)])).digest("hex");
  }
  const boundary = 2 ** Math.floor(Math.log2(data.length - 1));
  return createHash("sha256").update(Buffer.concat([
    Buffer.from([1]), Buffer.from(referenceRoot(data.slice(0, boundary)), "hex"),
    Buffer.from(referenceRoot(data.slice(boundary)), "hex")
  ])).digest("hex");
}

const RAW32 = Array.from({ length: 64 }, (_unused, index) =>
  Uint8Array.from({ length: 32 }, (_unusedByte, offset) => (index * 67 + offset * 13) % 256));
const INDEPENDENT_ROOTS = Array.from({ length: 65 }, (_unused, size) => referenceRoot(RAW32.slice(0, size)));
const INDEPENDENT_LEAVES = RAW32.map((data) => referenceRoot([data]));

function changedHash(hash: string): string {
  return `${hash[0] === "0" ? "1" : "0"}${hash.slice(1)}`;
}

describe("RFC 9162 offline tree", () => {
  test("empty, leaf and node hashes preserve byte domains and child order", () => {
    expect(RFC9162_EMPTY_ROOT).toBe(VENDOR_ROOTS[0]);
    expect(leafHash(new Uint8Array())).toBe(VENDOR_ROOTS[1]);
    const left = INDEPENDENT_LEAVES[0]!;
    const right = INDEPENDENT_LEAVES[1]!;
    const expectedNode = createHash("sha256").update(Buffer.concat([
      Buffer.from([1]), Buffer.from(left, "hex"), Buffer.from(right, "hex")
    ])).digest("hex");
    expect(nodeHash(left, right)).toBe(expectedNode);
    expect(leafHash(RAW32[0]!)).toBe(left);
    expect(leafHash(Buffer.from(Buffer.from(RAW32[0]!).toString("hex"), "utf8"))).not.toBe(left);
    expect(leafHash(Buffer.concat([Buffer.from(left, "hex"), Buffer.from(right, "hex")]))).not.toBe(expectedNode);
    expect(nodeHash(right, left)).not.toBe(expectedNode);
    expect(leafHash(new Uint8Array())).not.toBe(RFC9162_EMPTY_ROOT);
  });

  test("ports eight leaf inputs and all nine known roots from constants.go", () => {
    const data = VENDOR_LEAF_INPUTS.map((hex) => Buffer.from(hex, "hex"));
    const leaves = data.map((bytes) => leafHash(bytes));
    expect(data).toHaveLength(8);
    expect(VENDOR_ROOTS).toHaveLength(9);
    for (let size = 0; size <= 8; size += 1) {
      expect(rootHash(leaves.slice(0, size)), `vendor root at ${size}`).toBe(VENDOR_ROOTS[size]);
      expect(referenceRoot(data.slice(0, size)), `independent vendor root at ${size}`).toBe(VENDOR_ROOTS[size]);
    }
  });

  test("ports the five exact happy-path inclusion JSON vectors", () => {
    const leaves = VENDOR_LEAF_INPUTS.map((hex) => leafHash(Buffer.from(hex, "hex")));
    expect(VENDOR_INCLUSION).toHaveLength(5);
    for (const vector of VENDOR_INCLUSION) {
      const proof = (vector.proof ?? []).map(fromBase64);
      const root = fromBase64(vector.root);
      const leaf = fromBase64(vector.leafHash);
      expect(vector.wantErr, vector.sourcePath).toBe(false);
      expect(leaves[vector.leafIdx], vector.sourcePath).toBe(leaf);
      expect(rootHash(leaves.slice(0, vector.treeSize)), vector.sourcePath).toBe(root);
      expect(inclusionProof(leaves.slice(0, vector.treeSize), vector.leafIdx), vector.sourcePath).toEqual(proof);
      expect(verifyInclusion({ leafHash: leaf, leafIndex: vector.leafIdx, treeSize: vector.treeSize, proof, rootHash: root }), vector.sourcePath).toBe(true);
    }
  });

  test("ports the five exact happy-path consistency JSON vectors", () => {
    const leaves = VENDOR_LEAF_INPUTS.map((hex) => leafHash(Buffer.from(hex, "hex")));
    expect(VENDOR_CONSISTENCY).toHaveLength(5);
    for (const vector of VENDOR_CONSISTENCY) {
      const proof = (vector.proof ?? []).map(fromBase64);
      expect(vector.wantErr, vector.sourcePath).toBe(false);
      expect(rootHash(leaves.slice(0, vector.size1)), vector.sourcePath).toBe(fromBase64(vector.root1));
      expect(rootHash(leaves.slice(0, vector.size2)), vector.sourcePath).toBe(fromBase64(vector.root2));
      expect(consistencyProof(leaves.slice(0, vector.size2), vector.size1), vector.sourcePath).toEqual(proof);
      expect(verifyConsistency({ oldSize: vector.size1, newSize: vector.size2, oldRoot: fromBase64(vector.root1), newRoot: fromBase64(vector.root2), proof }), vector.sourcePath).toBe(true);
    }
  });

  test("matches an independent recursive root at every size from 1 through 64", () => {
    for (let size = 1; size <= 64; size += 1) {
      expect(rootHash(INDEPENDENT_LEAVES.slice(0, size)), `root at ${size}`).toBe(INDEPENDENT_ROOTS[size]);
    }
  });

  test("verifies every inclusion index against independent roots for sizes 1 through 64", () => {
    for (let size = 1; size <= 64; size += 1) {
      const leaves = INDEPENDENT_LEAVES.slice(0, size);
      for (let index = 0; index < size; index += 1) {
        expect(verifyInclusion({ leafHash: leaves[index]!, leafIndex: index, treeSize: size,
          proof: inclusionProof(leaves, index), rootHash: INDEPENDENT_ROOTS[size]! }), `index ${index}, size ${size}`).toBe(true);
      }
    }
  });

  test("verifies every consistency pair through size 64 against independent roots", () => {
    for (let newSize = 1; newSize <= 64; newSize += 1) {
      const leaves = INDEPENDENT_LEAVES.slice(0, newSize);
      for (let oldSize = 0; oldSize <= newSize; oldSize += 1) {
        expect(verifyConsistency({ oldSize, newSize, oldRoot: INDEPENDENT_ROOTS[oldSize]!,
          newRoot: INDEPENDENT_ROOTS[newSize]!, proof: consistencyProof(leaves, oldSize) }), `${oldSize} to ${newSize}`).toBe(true);
      }
    }
  });

  test("incremental and rebuilt frontiers match independent RFC roots through size 64", () => {
    const entryHashes = RAW32.map((bytes) => Buffer.from(bytes).toString("hex"));
    let frontier: MerkleFrontier = EMPTY_MERKLE_FRONTIER;
    for (let size = 0; size <= 64; size += 1) {
      expect(frontierRoot(frontier, "rfc9162-sha256"), `incremental at ${size}`).toBe(INDEPENDENT_ROOTS[size]);
      expect(frontierRoot(buildFrontierFromEntryHashes(entryHashes.slice(0, size), "rfc9162-sha256"), "rfc9162-sha256"), `rebuilt at ${size}`).toBe(INDEPENDENT_ROOTS[size]);
      expect(frontierMatchesLeafCount(frontier, size), `shape at ${size}`).toBe(true);
      if (size < 64) frontier = appendLeafToFrontier(frontier, entryHashes[size]!, "rfc9162-sha256");
    }
  });

  test("refuses the exact size-2 inclusion witness when only declared size changes to 3", () => {
    const leaves = INDEPENDENT_LEAVES.slice(0, 2);
    const input = { leafHash: leaves[0]!, leafIndex: 0, treeSize: 2,
      proof: inclusionProof(leaves, 0), rootHash: INDEPENDENT_ROOTS[2]! };
    expect(input.proof).toEqual([leaves[1]]);
    expect(verifyInclusion(input)).toBe(true);
    expect(verifyInclusion({ ...input, treeSize: 3 })).toBe(false);
  });

  test("rejects malformed, changed, extra, missing and wrong-index inclusion inputs", () => {
    const leaves = INDEPENDENT_LEAVES.slice(0, 8);
    const input = { leafHash: leaves[0]!, leafIndex: 0, treeSize: 8,
      proof: inclusionProof(leaves, 0), rootHash: INDEPENDENT_ROOTS[8]! };
    expect(verifyInclusion(input)).toBe(true);
    for (let offset = 0; offset < input.proof.length; offset += 1) {
      expect(verifyInclusion({ ...input, proof: input.proof.map((hash, index) => index === offset ? changedHash(hash) : hash) }), `tampered sibling ${offset}`).toBe(false);
    }
    expect(verifyInclusion({ ...input, proof: input.proof.slice(0, -1) })).toBe(false);
    expect(verifyInclusion({ ...input, proof: [...input.proof, leaves[7]!] })).toBe(false);
    expect(verifyInclusion({ ...input, leafHash: changedHash(input.leafHash) })).toBe(false);
    expect(verifyInclusion({ ...input, rootHash: changedHash(input.rootHash) })).toBe(false);
    expect(verifyInclusion({ ...input, leafIndex: 1 })).toBe(false);
    for (const invalid of ["", "00", "g".repeat(64), "A".repeat(64)]) {
      expect(verifyInclusion({ ...input, leafHash: invalid }), `leaf ${invalid}`).toBe(false);
      expect(verifyInclusion({ ...input, rootHash: invalid }), `root ${invalid}`).toBe(false);
      expect(verifyInclusion({ ...input, proof: [invalid, ...input.proof.slice(1)] }), `sibling ${invalid}`).toBe(false);
    }
    for (const leafIndex of [-1, 8, 0.5, NaN, Infinity, Number.MAX_SAFE_INTEGER + 1]) {
      expect(verifyInclusion({ ...input, leafIndex }), `index ${leafIndex}`).toBe(false);
      expect(() => inclusionProof(leaves, leafIndex)).toThrow();
    }
    for (const treeSize of [0, -1, 0.5, NaN, Infinity, Number.MAX_SAFE_INTEGER + 1]) {
      expect(verifyInclusion({ ...input, treeSize }), `size ${treeSize}`).toBe(false);
    }
    expect(() => inclusionProof([], 0)).toThrow();
  });

  test("rejects malformed consistency proofs, changed roots, invalid sizes and boundary mismatches", () => {
    const leaves = INDEPENDENT_LEAVES.slice(0, 8);
    const input = { oldSize: 3, newSize: 8, oldRoot: INDEPENDENT_ROOTS[3]!,
      newRoot: INDEPENDENT_ROOTS[8]!, proof: consistencyProof(leaves, 3) };
    expect(verifyConsistency(input)).toBe(true);
    for (let offset = 0; offset < input.proof.length; offset += 1) {
      expect(verifyConsistency({ ...input, proof: input.proof.map((hash, index) => index === offset ? changedHash(hash) : hash) }), `tampered sibling ${offset}`).toBe(false);
    }
    expect(verifyConsistency({ ...input, proof: input.proof.slice(0, -1) })).toBe(false);
    expect(verifyConsistency({ ...input, proof: [...input.proof, leaves[7]!] })).toBe(false);
    expect(verifyConsistency({ ...input, oldRoot: changedHash(input.oldRoot) })).toBe(false);
    expect(verifyConsistency({ ...input, newRoot: changedHash(input.newRoot) })).toBe(false);
    for (const invalid of ["", "00", "g".repeat(64), "A".repeat(64)]) {
      expect(verifyConsistency({ ...input, oldRoot: invalid })).toBe(false);
      expect(verifyConsistency({ ...input, newRoot: invalid })).toBe(false);
      expect(verifyConsistency({ ...input, proof: [invalid, ...input.proof.slice(1)] })).toBe(false);
    }
    for (const [oldSize, newSize] of [[-1, 8], [3, 2], [1.5, 8], [1, Infinity], [NaN, 8], [0, Number.MAX_SAFE_INTEGER + 1]]) {
      expect(verifyConsistency({ ...input, oldSize, newSize }), `sizes ${oldSize}, ${newSize}`).toBe(false);
    }
    for (const oldSize of [-1, 9, 0.5, NaN, Infinity, Number.MAX_SAFE_INTEGER + 1]) {
      expect(() => consistencyProof(leaves, oldSize)).toThrow();
    }
    const equal = { ...input, oldSize: 8, proof: [] };
    expect(verifyConsistency({ ...equal, oldRoot: input.newRoot })).toBe(true);
    expect(verifyConsistency({ ...equal, oldRoot: input.newRoot, newRoot: changedHash(input.newRoot) })).toBe(false);
    expect(verifyConsistency({ ...equal, oldRoot: input.newRoot, proof: [leaves[0]!] })).toBe(false);
    const empty = { ...input, oldSize: 0, oldRoot: RFC9162_EMPTY_ROOT, proof: [] };
    expect(verifyConsistency(empty)).toBe(true);
    expect(verifyConsistency({ ...empty, oldRoot: changedHash(RFC9162_EMPTY_ROOT) })).toBe(false);
    expect(verifyConsistency({ ...empty, proof: [leaves[0]!] })).toBe(false);
    // No stronger zero/zero root-canonicality contract is inferred here.
  });
});
